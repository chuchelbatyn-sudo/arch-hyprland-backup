Соответствие путей: backup -> исходный путь

home/.config/*              -> ~/.config/*
home/.gtkrc-2.0             -> ~/.gtkrc-2.0
home/.local/bin/*           -> ~/.local/bin/*
home/.local/share/applications/* -> ~/.local/share/applications/*
system/etc/*                -> /etc/*
system/boot/grub.cfg        -> /boot/grub/grub.cfg
gtk/*                       -> ~/.config/{gtk-3.0,gtk-4.0,xsettingsd,nwg-look}, ~/.gtkrc-2.0
applications/*              -> ~/.config/PortProton.conf, ~/.config/Gaijin
